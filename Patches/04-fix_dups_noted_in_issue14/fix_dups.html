The question then is do we need separate shapes for the 98th and the 99th through 102nd. The answer is yes. The 98th districts are different:

nydat |>
  mutate(area.1 = as.numeric(st_area(geometry))) |>
  st_intersection(nydat) |> 
  glimpse() |>
  filter(fn != fn.1) |> 
  mutate(over_area = as.numeric(st_area(geometry)),
         overlap = over_area/area.1) |> 
  filter(overlap > 0.01,
         fn %in% c("098_to_098"),
         fn.1 %in% c("099_to_102")) |>
  select(fn, fn.1, district, district.1, overlap) |> 
  st_drop_geometry() |>
  arrange(district) |> 
  gt::gt()
